<script>{`
  // DOM取得
  const fswInput = document.getElementById('fsw-input');
  const sandboxPreview = document.getElementById('sandbox-preview');
  const copyBtn = document.getElementById('copy-btn');
  const presetWords = document.querySelectorAll('.preset-word');
  const presetSymbols = document.querySelectorAll('.preset-symbol');

  // 入力監視とリアルタイム更新
  fswInput.addEventListener('input', (e) => {
    const code = e.target.value.trim();
    if (code) {
      sandboxPreview.setAttribute('fsw', code);
    }
  });

  // 単語プリセット処理
  presetWords.forEach(button => {
    button.addEventListener('click', () => {
      const fsw = button.getAttribute('data-fsw');
      fswInput.value = fsw;
      sandboxPreview.setAttribute('fsw', fsw);
      
      // クリックフィードバック用アニメーション
      button.style.transform = 'scale(0.95)';
      setTimeout(() => button.style.transform = 'scale(1)', 100);
    });
  });

  // シンボルプリセット処理
  presetSymbols.forEach(button => {
    button.addEventListener('click', () => {
      const fsw = button.getAttribute('data-fsw');
      fswInput.value = fsw;
      sandboxPreview.setAttribute('fsw', fsw);
      
      button.style.transform = 'scale(0.95)';
      setTimeout(() => button.style.transform = 'scale(1)', 100);
    });
  });

  // コピー処理
  copyBtn.addEventListener('click', () => {
    navigator.clipboard.writeText(fswInput.value).then(() => {
      const originalText = copyBtn.innerText;
      copyBtn.innerText = 'コピー完了！';
      copyBtn.style.background = 'linear-gradient(135deg, #10b981, #059669)'; // 緑色グラデーション
      
      setTimeout(() => {
        copyBtn.innerText = originalText;
        copyBtn.style.background = ''; // デフォルトに戻す
      }, 2000);
    }).catch(err => {
      console.error('コピー失敗:', err);
      alert('コピーに失敗しました。お手数ですが手動で選択してコピーしてください。');
    });
  });
`}</script>
